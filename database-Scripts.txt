CREATE TABLE Department
(
    DeptID INT PRIMARY KEY,
    DeptName NVARCHAR(100) NOT NULL
);



CREATE TABLE Employee
(
    EmpID INT IDENTITY(1001,1) PRIMARY KEY,

    EmpName NVARCHAR(100) NOT NULL,

    DeptID INT NOT NULL,

    IsActive BIT DEFAULT 1,

    FOREIGN KEY(DeptID)
    REFERENCES Department(DeptID)
);



CREATE TABLE EmployeeLogin
(
    LoginID INT IDENTITY(1,1) PRIMARY KEY,

    EmpID INT NOT NULL UNIQUE,

    EmpCode NVARCHAR(50) UNIQUE NOT NULL,

    PasswordHash NVARCHAR(100) NOT NULL,

    Role NVARCHAR(20) NOT NULL,

    IsActive BIT DEFAULT 1,

    FOREIGN KEY(EmpID)
    REFERENCES Employee(EmpID)
    ON DELETE CASCADE
);



CREATE TABLE Attendance
(
    AttendanceID INT IDENTITY(1,1) PRIMARY KEY,

    EmpID INT NOT NULL,

    AttendanceDate DATE NOT NULL,

    Status NVARCHAR(20) NOT NULL,

    FOREIGN KEY(EmpID)
    REFERENCES Employee(EmpID)
    ON DELETE CASCADE
);



CREATE TABLE LeaveRequest
(
    LeaveID INT IDENTITY(1,1) PRIMARY KEY,

    EmpID INT NOT NULL,

    LeaveType NVARCHAR(30),

    FromDate DATE,

    ToDate DATE,

    Reason NVARCHAR(200),

    Status NVARCHAR(20),

    FOREIGN KEY(EmpID)
    REFERENCES Employee(EmpID)
    ON DELETE CASCADE
);



CREATE TABLE Salary
(
    SalaryID INT IDENTITY(1,1) PRIMARY KEY,

    EmpID INT NOT NULL,

    BasicSalary DECIMAL(10,2),

    Bonus DECIMAL(10,2),

    Deduction DECIMAL(10,2),

    NetSalary AS (BasicSalary + Bonus - Deduction),

    FOREIGN KEY(EmpID)
    REFERENCES Employee(EmpID)
    ON DELETE CASCADE
);

CREATE PROCEDURE sp_GetAllEmployees
AS
BEGIN
    SELECT
        E.EmpID,
        E.EmpName,
        E.Salary,
        D.DeptName,
        E.IsActive
    FROM Employee E
    INNER JOIN Department D
        ON E.DeptID = D.DeptID;
END;
GO

    @EmpID INT

BEGIN

CREATE PROCEDURE sp_UpdateEmployee
(
    @EmpID INT,
    @EmpName NVARCHAR(100),
    @Salary DECIMAL(10,2),
    @DeptID INT
)
AS
BEGIN
    UPDATE Employee

    SET
        EmpName=@EmpName,
        Salary=@Salary,
        DeptID=@DeptID

    WHERE EmpID=@EmpID;
END;
GO

CREATE PROCEDURE sp_Login
(
    @EmpCode NVARCHAR(20),
    @Password NVARCHAR(100)
)
AS
BEGIN

    SELECT *

    FROM EmployeeLogin

    WHERE
        EmpCode=@EmpCode
        AND PasswordHash=@Password
        AND IsActive= 1;

END;
GO

    Description NVARCHAR(300)

CREATE TRIGGER trg_Employee_Insert
ON Employee
AFTER INSERT
AS
BEGIN

    INSERT INTO EmployeeAudit
    (
        EmpID,
        ActionType,
        Description
    )

    SELECT

        EmpID,

        'INSERT',

        'New Employee Added : ' + EmpName

    FROM inserted;

END;
GO

    INSERT INTO EmployeeAudit

        'Employee Details Updated'

    FROM inserted;

CREATE TRIGGER trg_Employee_Delete
ON Employee
AFTER DELETE
AS
BEGIN

    INSERT INTO EmployeeAudit
    (
        EmpID,
        ActionType,
        Description
    )

    SELECT

        EmpID,

        'DELETE',

        'Employee Deleted'

    FROM deleted;

END;
GO

CREATE VIEW vw_EmployeeDetails
AS

SELECT

E.EmpID,

E.EmpName,

D.DeptName,

E.Salary,

E.IsActive

FROM Employee E

INNER JOIN Department D

ON E.DeptID=D.DeptID;
GO

--
go
CREATE VIEW vw_AttendanceReport
AS

SELECT

E.EmpName,

A.AttendanceDate,

A.Status

FROM Employee E

INNER JOIN Attendance A

ON E.EmpID=A.EmpID;
GO

--
go
CREATE VIEW vw_SalaryReport
AS

SELECT

E.EmpName,

S.BasicSalary,

S.Bonus,

S.Deduction,

S.NetSalary

FROM Employee E

INNER JOIN Salary S

ON E.EmpID=S.EmpID;
GO
